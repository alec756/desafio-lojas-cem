import { ProductService } from "../services/product.service";

export class ProductController {
    private service: ProductService;

    constructor() {
        this.service = new ProductService();
    }

    async create(request: any, reply: any) {
        try {
            const { name, description, price, stock_quantity } = request.body;
            const product = await this.service.createProduct({
                name, description, price, stock_quantity,
            });
            return reply.status(201).send(product);
        } catch (error: any) {
            return reply.status(400).send({ error: error.message });
        }
    }

    async index(request: any, reply: any) {
        const products = await this.service.listProducts();
        return reply.send(products);
    }
}
